cmake_minimum_required(VERSION 3.16)
project(component_host_tests C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_library(component STATIC src/component.c)
target_include_directories(component PUBLIC include)
add_executable(test_component tests/test_component.c)
target_link_libraries(test_component PRIVATE component)
add_test(NAME component COMMAND test_component)
